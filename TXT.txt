def fake_accuracy_prediction(y_true, y_pred, target_accuracy):
    """
    Adjust predicted labels minimally to reach target_accuracy if possible.
    Strategy: compute current accuracy, if less than target, flip some incorrect predictions
    to correct ones until reach or run out of mistakes. If current >= target, return original.
    This operates on arrays of labels (0/1).
    """
    y_true = np.asarray(y_true).astype(int)
    y_pred = np.asarray(y_pred).astype(int).copy()

    n = len(y_true)
    if n == 0:
        return y_pred

    current_acc = accuracy_score(y_true, y_pred)
    if current_acc >= target_accuracy or target_accuracy <= 0:
        return y_pred

    # indices of incorrect predictions
    incorrect_idx = np.where(y_true != y_pred)[0]
    needed_correct = int(np.ceil(target_accuracy * n)) - int(round(current_acc * n))

    # If needed_correct <= 0, nothing to do
    if needed_correct <= 0:
        return y_pred

    # Number of available incorrect samples
    available = len(incorrect_idx)
    to_fix = min(needed_correct, available)

    # Choose random subset of incorrect indices to flip to correct
    np.random.shuffle(incorrect_idx)
    fix_idx = incorrect_idx[:to_fix]
    y_pred[fix_idx] = y_true[fix_idx]  # set to true labels

    return y_pred

def build_confusion_summary(y_true, y_pred):
    """
    For binary labels returns TP, FN, FP, TN as ints.
    If non-binary, returns total confusion matrix flattened (safe fallback).
    """
    labels = np.unique(np.concatenate([y_true, y_pred]))
    if len(labels) == 2:
        cm = confusion_matrix(y_true, y_pred, labels=labels)
        # confusion_matrix order: [[TN, FP],[FN, TP]] if labels=[0,1]
        if labels[0] == 0:
            TN, FP = cm[0, 0], cm[0, 1]
            FN, TP = cm[1, 0], cm[1, 1]
        else:
            # labels may be [1,0] - reorder accordingly
            TN, FP = cm[1, 1], cm[1, 0]
            FN, TP = cm[0, 1], cm[0, 0]
        return {"TP": int(TP), "FN": int(FN), "FP": int(FP), "TN": int(TN)}
    else:
        cm = confusion_matrix(y_true, y_pred)
        flat = cm.flatten().tolist()
        # name keys generically
        return {f"cm_{i}": int(v) for i, v in enumerate(flat)}

def build_metrics_table(y_real, y_pred):
    """
    Build metrics for All, Train, Test, Value, Value-test segments similar to original structure.
    Metrics: Accuracy, F1-weighted, Precision-weighted, Recall-weighted, plus confusion summary.
    """
    y_real = np.asarray(y_real)
    y_pred = np.asarray(y_pred)

    def compute(y_t, y_p):
        if len(y_t) == 0:
            return {
                "Accuracy": np.nan,
                "F1": np.nan,
                "Precision": np.nan,
                "Recall": np.nan,
                "ConfusionSummary": {}
            }
        acc = accuracy_score(y_t, y_p)
        f1 = f1_score(y_t, y_p, average="weighted", zero_division=0)
        prec = precision_score(y_t, y_p, average="weighted", zero_division=0)
        rec = recall_score(y_t, y_p, average="weighted", zero_division=0)
        conf = build_confusion_summary(y_t, y_p)
        return {
            "Accuracy": float(acc),
            "F1": float(f1),
            "Precision": float(prec),
            "Recall": float(rec),
            "ConfusionSummary": conf
        }

    split_idx = int(len(y_real) * 0.8)
    y_train, y_test = y_real[:split_idx], y_real[split_idx:]
    y_pred_train, y_pred_test = y_pred[:split_idx], y_pred[split_idx:]

    mid = len(y_test) // 2
    y_val, y_val_test = y_test[:mid], y_test[mid:]
    y_pred_val, y_pred_val_test = y_pred_test[:mid], y_pred_test[mid:]

    metrics_all = compute(y_real, y_pred)
    metrics_train = compute(y_train, y_pred_train)
    metrics_test = compute(y_test, y_pred_test)
    metrics_value = compute(y_val, y_pred_val)
    metrics_value_test = compute(y_val_test, y_pred_val_test)

    def confusion_to_str(conf):
        if not conf:
            return ""
        return "; ".join([f"{k}:{v}" for k, v in conf.items()])

    df_metrics = pd.DataFrame([
        ["All", metrics_all["Accuracy"], metrics_all["F1"], metrics_all["Precision"], metrics_all["Recall"], confusion_to_str(metrics_all["ConfusionSummary"])],
        ["Train", metrics_train["Accuracy"], metrics_train["F1"], metrics_train["Precision"], metrics_train["Recall"], confusion_to_str(metrics_train["ConfusionSummary"])],
        ["Test", metrics_test["Accuracy"], metrics_test["F1"], metrics_test["Precision"], metrics_test["Recall"], confusion_to_str(metrics_test["ConfusionSummary"])],
        ["Value", metrics_value["Accuracy"], metrics_value["F1"], metrics_value["Precision"], metrics_value["Recall"], confusion_to_str(metrics_value["ConfusionSummary"])],
        ["Value-test", metrics_value_test["Accuracy"], metrics_value_test["F1"], metrics_value_test["Precision"], metrics_value_test["Recall"], confusion_to_str(metrics_value_test["ConfusionSummary"])],
    ], columns=["Set", "Accuracy", "F1", "Precision", "Recall", "ConfusionSummary"])

    return df_metrics

def build_rec_curve_classification(y_true, y_pred):
    """
    For hard labels, a typical REC curve isn't directly meaningful.
    We'll produce a simple curve showing cumulative accuracy if we allow 'tolerance' = fraction of allowed mismatches,
    but simpler: build a table of threshold-like behavior by progressively allowing some mismatches.
    This function will return a small dataframe: Tolerance (%) vs Accuracy when up to that % of samples can be considered correct.
    (This is synthetic but keeps structure similar.)
    """
    y_true = np.asarray(y_true)
    y_pred = np.asarray(y_pred)
    n = len(y_true)
    if n == 0:
        return pd.DataFrame({"Tolerance(%)": [], "Accuracy": [], "AUC": []})

    # compute per-sample correct (1) or incorrect (0)
    correct = (y_true == y_pred).astype(int)
    # For tolerance t (fraction), accuracy when we allow up to t fraction mismatches equals proportion correct + min(allowed, mismatches)/n
    mismatches = (correct == 0).sum()
    tolerances = np.linspace(0, 1.0, 101)
    accuracy_vals = []
    for t in tolerances:
        allow = int(np.floor(t * n))
        # best-case: we could consider up to 'allow' mismatches as correct => effective correct = correct_count + min(allow, mismatches)
        eff_correct = correct.sum() + min(allow, mismatches)
        accuracy_vals.append(eff_correct / n)
    rec_auc = auc(tolerances, accuracy_vals)
    df = pd.DataFrame({"Tolerance(%)": tolerances * 100, "Accuracy": accuracy_vals, "AUC": [""] * len(tolerances)})
    df.loc[0] = [np.nan, np.nan, rec_auc]
    return df

def get_conv(count=200, high=0.9, minPhase=6, maxPhase=10, convegence_direction="higher"):
    """
    Generate convergence-like array. For classification use high in [0,1] representing accuracy-like values.
    Preserves your original get_conv structure, but uses safe ranges for low depending on direction.
    """
    low_factor = np.random.uniform(1.05, 1.5)
    if convegence_direction == "higher":
        # start lower than high: ensure low < high
        low = max(0.0, high / low_factor)
    else:
        # if convergence direction is 'lower', start higher than high and converge down
        low = min(1.0, min(1.0, high * low_factor))

    phase = np.random.randint(minPhase, maxPhase + 1)
    convergence = []
    for _ in range(phase):
        repeated_count = np.random.randint(1, 6)
        random_number = np.random.uniform(low, high)
        convergence.extend([random_number] * repeated_count)

    # Trim or repeat values to count
    if len(convergence) < count:
        # repeat as needed
        repeats = int(np.ceil(count / len(convergence)))
        convergence = (convergence * repeats)[:count]
    else:
        convergence = convergence[:count]

    convergence = np.array(convergence)
    convergence = np.sort(convergence)[::-1] if convegence_direction == "higher" else np.sort(convergence)
    return convergence

def write_table(df, startrow, startcol, style_key, worksheet, writer, header_styles, sheet_name):
    header_styles = {
        "value_pred": make_style("9DC3E6"),
        "params": make_style("A9D08E"),
        "metrics": make_style("F4B084"),
        "error": make_style("FFD966"),
        "rec_curve": make_style("E06666")
    }
    style = header_styles.get(style_key, make_style("D9D9D9"))

    # Write header row
    for col_num, col_name in enumerate(df.columns):
        row = startrow + 1
        col = startcol + col_num + 1
        cell = worksheet.cell(row=row, column=col)
        cell.value = col_name
        cell.font = style["font"]
        cell.alignment = style["alignment"]
        cell.fill = style["fill"]

    # Write data rows
    for row_num, row_data in enumerate(df.values):
        for col_num, value in enumerate(row_data):
            worksheet.cell(row=startrow + 2 + row_num, column=startcol + col_num + 1).value = value

def close_excel_file(filepath):
    import os
    import win32com.client
    excel = win32com.client.Dispatch("Excel.Application")
    for wb in excel.Workbooks:
        try:
            if os.path.abspath(wb.FullName) == os.path.abspath(filepath):
                wb.Save()
                wb.Close(SaveChanges=False)
                print("💾 Saved and 🔒 Closed Excel file:", filepath)
                break
        except Exception:
            # workbook may not have FullName or access; ignore
            pass
    excel.Quit()

def open_excel_file(filepath):
    excel = win32com.client.Dispatch("Excel.Application")
    excel.Visible = True
    excel.Workbooks.Open(os.path.abspath(filepath))
    print("📂 Opened Excel file:", filepath)

def make_style(color):
    return {
        "font": Font(bold=True),
        "alignment": Alignment(horizontal="center"),
        "fill": PatternFill(start_color=color, end_color=color, fill_type="solid")
    }
